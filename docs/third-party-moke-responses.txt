    public function generateRequest(ClientServiceRequest $request)
    {
        try {
            $service = Service::with(['apis' => function ($query) use ($request) {
                $query->where('name', $request->api);
            }])->where('name', $request->service)->first();
            $apiId = optional($service->apis->first())->id;
            $access = $this->authenticateAccess($request->client_id, $service, $apiId);
            if ($access) {
                $request->merge(['service_id' => $service->id]);
                $request->merge(['product_id' => 1]);
                $request->merge(['api_id' => $apiId]);
                $client_request = $this->clientRequestService->generateClientRequest($request);
                Log::info($request->service . " against request " . $client_request->request_id);
                $client = Client::find($request->client_id);
                $request->merge(['client' => $client]);
                $apiEnviornment = ApiEnvironment::where('api_id', $apiId)->where('env', $client->env)->first();
                $request->merge(['api_enviornment' => $apiEnviornment]);
                if ($client->env == "TEST") {
                    $request_response = GET_MOCK_RESPONSE($request);
                    $response = $request_response;
                    $status = 200;
                } else {
                    switch ($request->service) {
                        case 'YAKEEN':
                            if ($request->api === 'COMPANY_ADDRESS') {
                                $request_response = $this->clientRequestService->companyAddressRequest($request);
                            } else {
                                $request_response = $this->clientRequestService->kycRequest($request);
                            }
                            break;
                        case 'YAKEEN LITE':
                            $request_response = $this->clientRequestService->mobileVerificationRequest($request);
                            break;
                        case 'BAYAN':
                            if ($request->api == 'BAYAN_ME_FINANCIAL') {
                                $request_response = $this->clientRequestService->bayanFinancialRequest($request);
                            } elseif ($request->api == 'BAYAN_CREDIT') {
                                $request_response = $this->clientRequestService->bayanCreditRequest($request);
                            } else {
                                $request_response = $this->clientRequestService->bayanNaeRequest($request);
                            }
                            break;
                        case 'SIMAH':
                            if ($request->api == 'SIMAH_CONSUMER') {
                                $request_response = $this->clientRequestService->simahConsumerRequest($request);
                            } else {
                                $request_response = $this->clientRequestService->simahConsumerV2Request($request);
                            }
                            break;
                        case 'NAFATH':
                            $request_response = $this->clientRequestService->nafathRequest($request);
                            break;
                        case 'ORBIT SMS':
                            $request_response = $this->clientRequestService->orbitSMSRequest($request);
                            break;
                        case 'AML WIZARD':
                            $request_response = $this->clientRequestService->amlWizardRequest($request);
                            break;
                        case 'DHAMEN':
                            $request_response = $this->clientRequestService->ibanValidationRequest($request);
                            break;
                        case 'COMMODITY':
                            if ($request->api === 'NEW_ORDER') {
                                $request_response = $this->clientRequestService->commodityNewOrderRequest($request);
                            } elseif ($request->api === 'MATCHING_RESULT') {
                                $request_response = $this->clientRequestService->commodityMatchingResultRequest($request);
                            } else {
                                $request_response = $this->failure('Invalid API Name for Commodity Service!');
                            }
                            break;
                        case 'WATHQ':
                            $request_response = $this->clientRequestService->wathqRequest($request);
                            break;
                        case 'SIGNIT':
                            $request_response = $this->clientRequestService->signitRequest($request);
                            break;
                        default:
                            $request_response = $this->failure('Invalid Service Name !');
                            break;
                    }

                    $response = json_encode($request_response->getData());
                    $status = $request_response->getStatusCode();
                }
                if ($request->service == "NAFITH") {
                    $getNafithData = $request_response->getData();
                    if ($getNafithData->data->httpcode == 200 && isset($getNafithData->data->response->data)) {
                        $client_request->contract_number = $getNafithData->data->response->data->contractNumber;
                    }
                }
//                $client_request->request = json_encode($request->all());
                $client_request->responses = $response;
                $client_request->status = $status;
                $client_request->save();
                return $request_response;
            } else {
                return $this->unauthorizedAccess('Authorization Failed');
            }
        } catch (Exception $exception) {
            return $this->failure(
                $exception->getMessage(),
                $exception->getCode()
            );
        }
    }



    public function mobileVerificationRequest($request)
    {
        $request->merge(['iqama_id' => $request->nid, 'mobile_no' => $request->mobile_number]);
        $array = CURL_REQUEST_MOBILE_VERIFICATION($request);
        Log::info("Mobile Verification Request against Nid " . $request->nid, ["Response" => $array]);

        // dd($array);
        if ($array) {
            if (isset($array['isOwner'])) {
                // dd("dds");
                return $this->success('Request Generated', $array);
            } else if (isset($array['code'])) {
                return $this->failureWithMessage('', $array['message']);
            }
            return $this->failure('Something went wrong, please try again!');
        }
        return $this->failure('Something went wrong, please try again!');
    }


function CURL_REQUEST_MOBILE_VERIFICATION($request)
{
    $verification_method = GET_VERIFICATION_METHOD($request->product_id, $request->api_id);
    $curl = curl_init();

    curl_setopt_array($curl, array(
        CURLOPT_URL => $verification_method['url'] . $request->nid . '/owns-mobile/' . $request->mobile_no,
        CURLOPT_RETURNTRANSFER => true,
        CURLOPT_ENCODING => '',
        CURLOPT_MAXREDIRS => 10,
        CURLOPT_TIMEOUT => 0,
        CURLOPT_FOLLOWLOCATION => true,
        CURLOPT_HTTP_VERSION => CURL_HTTP_VERSION_1_1,
        CURLOPT_CUSTOMREQUEST => $verification_method['method'],
        CURLOPT_HTTPHEADER => array(
            'APP-ID: ' . $verification_method['credentials']['APP-ID'],
            'APP-KEY: ' . $verification_method['credentials']['APP-KEY'],
            'SERVICE_KEY: 884aa9bc-06dd-4750-a787-4a0418d9565e',
            'ORGANIZATION-NUMBER: 7001734636',
        ),
    ));

    $response = curl_exec($curl);

    curl_close($curl);
    Log::info("Mobile verification response:", ["body" => $response]);
    $headers = [
        'APP-ID: ' . $verification_method['credentials']['APP-ID'],
        'APP-KEY:' . $verification_method['credentials']['APP-KEY'],
        'SERVICE_KEY: ' . $verification_method['credentials']['SERVICE_KEY'],
        'ORGANIZATION-NUMBER: ' . $verification_method['credentials']['ORGANIZATION-NUMBER'],
    ];
    $requestData = null;
    $curlCommand = 'curl -X GET \'' . $verification_method['url'] . $request->nid . '/owns-mobile/' . $request->mobile_no . '\'';
    foreach ($headers as $header) {
        $curlCommand .= ' -H \'' . $header . '\'';
    }
    $curlCommand .= ' --data-raw \'' . $requestData . '\'';
    Log::info("MOBILE VERIFICATION CURL REQUEST:", ["body" => $curlCommand]);
    $response = shell_exec($curlCommand);
    $response = curl_exec($curl);
    curl_close($curl);

    return json_decode($response, true);
}


    public function simahConsumerV2Request($request)
    {
        if ($request->api == 'SIMAH_CONSUMER_V2') {
            $api_response = CURL_REQUEST_SIMAH_CONSUMER_V2($request);
            Log::info("Simah consumer version 2 Response", ["SIMAH_CONSUMER_V2" => $api_response]);
            if ($api_response) {
                if ($api_response['httpcode'] == 200 && $api_response['response']['isSuccess'] == true)
                    return $this->success('Success', $api_response['response']);
                else
                    return $this->failureWithMessage('Failure', $api_response['response']);
            } else {
                return $this->failureWithMessage('Something went wrong, please try again!');
            }
        } else {
            return $this->failureWithMessage('The requested API does not exist.');
        }
    }


function CURL_REQUEST_SIMAH_CONSUMER_V2($request)
{
    $verification_method = GET_SIMAH_CONSUMER_V2_VERIFICATION_METHOD('SIMAH_CONSUMER_V2', $request->client->env);

    $uniqueReferenceNumber = GeneralHelper::randomNumber();
    $request->merge([
        'referenceNumber' => $uniqueReferenceNumber,
        'productType' => 23,
        'accept' => true
    ]);
    $body = $request->except([
        'service_id',
        'product_id',
        'client_id',
        'client',
        'service',
        'api',
        'ip'
    ]);

    $url = $verification_method['url'];
    $endpoint = $verification_method['endpoint'];
    $method = strtoupper($verification_method['method'] ?? 'POST');

        $headers = [
            'Content-Type: application/json',
        'Accept: text/plain',
        'language: En',
            'Authorization: Bearer ' . $verification_method['authorization_token'],
        ];

    if ($verification_method != false) {

        $curl = curl_init();

        curl_setopt_array($curl, array(
            CURLOPT_URL => $url . $endpoint,
            CURLOPT_RETURNTRANSFER => true,
            CURLOPT_ENCODING => '',
            CURLOPT_MAXREDIRS => 10,
            CURLOPT_TIMEOUT => 0,
            CURLOPT_FOLLOWLOCATION => true,
            CURLOPT_HTTP_VERSION => CURL_HTTP_VERSION_1_1,
            CURLOPT_CUSTOMREQUEST => 'POST',
            CURLOPT_POSTFIELDS => json_encode($body),
            CURLOPT_HTTPHEADER => $headers,
        ));

        // ✅ Build the cURL command string
        $curlCommand = "curl -X POST '" . $url . $endpoint . "'";

        foreach ($headers as $header) {
            $curlCommand .= " -H \"" . $header . "\"";
        }
        $curlCommand .= " -d '" . json_encode($body) . "'";
        // 🔵 Log the generated cURL command
        Log::info('Generated SIMAH CONSUMER V2 CURL command:', ['command' => $curlCommand]);
        $response = curl_exec($curl);
        $httpcode = curl_getinfo($curl, CURLINFO_HTTP_CODE);
        curl_close($curl);
        return [
            'httpcode' => $httpcode,
            'response' => json_decode($response, true)
        ];
    } else {
        $response = [];
        $response['httpcode'] = 404;
        return $response;
    }
}


function GET_SIMAH_CONSUMER_V2_VERIFICATION_METHOD($name, $env)
{
    $login_verification_method = VerificationMethod::where('name', 'SIMAH_CONSUMER_LOGIN_V2')
        ->where('env', $env)
        ->first();
    $headers = $login_verification_method->headers ? json_decode($login_verification_method->headers, true) : [];
    $start = $now = Carbon::now('UTC');
    $end = isset($headers['last_updated'])
        ? Carbon::parse($headers['last_updated'])->setTimezone('UTC')
        : $start;
    $differenceInMinutes = $start->diffInMinutes($end);
    if (empty($headers['Authorization']) || $differenceInMinutes >= 30) {
        $response = GET_SIMAH_CONSUMER_V2_AUTHORIZATION_TOKEN($login_verification_method);
        Log::info("Simah consumer authorization token Request Data:", ["response" => $response]);
        if ($response['httpcode'] === 200 && $response['response']['isSuccess'] === true) {
            $headers['Authorization'] = $response['response']['data']['token'];
            $headers['last_updated'] = now();
            $login_verification_method->headers = json_encode($headers);
            $login_verification_method->save();
        }
    }

    $curl_request_details = VerificationMethod::where('name', $name)
        ->where('env', $env)
        ->first();
    return [
        'authorization_token' => $headers['Authorization'] ?? null,
        'x_company' => $headers['X-Company'] ?? null,
        'url' => $curl_request_details->url,
        'endpoint' => $curl_request_details->endpoint,
        'method' => $curl_request_details->method,
        'headers' => $headers,
        'credentials' => json_decode($curl_request_details->credentials, true),
    ];
}


function GET_SIMAH_CONSUMER_V2_AUTHORIZATION_TOKEN($login_verification_method)
{

    $url = $login_verification_method->url;
    $endpoint = $login_verification_method->endpoint;
    $method = strtoupper($login_verification_method->method ?? 'POST');
    $credentials = json_decode($login_verification_method->credentials, true);
    $fullUrl = $url . $endpoint;

    $postFields = $credentials; // Make credentials dynamic


    $curl = curl_init();

    curl_setopt_array($curl, array(
        CURLOPT_URL => $fullUrl,
        CURLOPT_RETURNTRANSFER => true,
        CURLOPT_ENCODING => '',
        CURLOPT_MAXREDIRS => 10,
        CURLOPT_TIMEOUT => 0,
        CURLOPT_FOLLOWLOCATION => true,
        CURLOPT_HTTP_VERSION => CURL_HTTP_VERSION_1_1,
        CURLOPT_CUSTOMREQUEST => 'POST',
        CURLOPT_POSTFIELDS => json_encode($postFields),
        CURLOPT_HTTPHEADER => array(
        'Content-Type: application/json',
            'Accept: application/json'
        ),
    ));
    // ✅ Generate equivalent CURL command string
    $curlCommand = "curl -X {$method} '{$fullUrl}'";
    $curlCommand .= " -H 'Content-Type: application/json'";
    $curlCommand .= " --data '" . json_encode($postFields) . "'";
    Log::info("Lean authorization token Request Data:", ["body" => $curlCommand]);

    // 🔵 Log the generated CURL command
    Log::info('Generated SIMAH CURL command for login verification:', ['command' => $curlCommand]);

    $response = curl_exec($curl);
    $httpcode = curl_getinfo($curl, CURLINFO_HTTP_CODE);

    curl_close($curl);


    return [
        'httpcode' => $httpcode,
        'response' => json_decode($response, true)
    ];
}
